Code Snippet


// import UltraSonic Distance Sensor libiary
#include 
// import Servo Motor libiary
#include 
// initialize UltraSonic Distance Sensor trigger pin as pin 9.
const int trigPin = 9;
// initialize UltraSonic Distance Sensor echo pin as pin 10.
const int echoPin = 10;
// initialize sensor that uses digital pins 9 and 10.
UltraSonicDistanceSensor distanceSensor(trigPin, echoPin);
// create Servo object to control a servo
Servo myservo;
// initialize servo on pin 6
const int servoPin = 6;
// value to input into the servo motor
int val;
// minimum sensor value
int sensorMin;
// maxim sensor value
int sensorMax;

void setup() {
  // attaches the servo to servo pin.
  myservo.attach(servoPin);
  // initialize serial connection to print values from sensor.
  Serial.begin(9600);
  // notify user that minimum value will be recorded in 2 seconds
  Serial.println("Recording minimum sensor value in 2 seconds");
  // waits 2 seconds to read the lowest value
  delay(2000);
  // check for the lowest value you get
  sensorMin = distanceSensor.measureDistanceCm();
  // notify user the recorded sensorMin value
  Serial.print("Record minimum value successful, sensorMin is ");
  Serial.println(sensorMin);
  // notify user that maxim value will be recorded in 2 seconds
  Serial.println("Recording maxim sensor value in 2 seconds");
  // waits for 2000 ms to read the higest value
  delay(2000);
  // check for the highest value you get
  sensorMax = distanceSensor.measureDistanceCm();
  // notify user the recorded sensorMax value
  Serial.print("Record maxim value successful, sensorMax is ");
  Serial.println(sensorMax);
}

void loop() {
  // do a measurement using the sensor and record distance in centimeters.
  val = distanceSensor.measureDistanceCm();
  // check to see if distance value is greater than recorded maxim sensor value
  if (val > sensorMax) {
    // update maxim sensor value to the new value
    sensorMax = val;
    // print message to notify user the new maxim sensor value
    Serial.print("sensorMax updated to ");
    Serial.println(sensorMax);
  }
  // check to see if distance value is greater than recorded minimum sensor value
  if (val < sensorMin) {
    // update minimum sensor value to the new value
    sensorMin = val;
    // print message to notify user the new minimum sensor value
    Serial.print("sensorMin updated to ");
    Serial.println(sensorMin);
  }
  // scale the value for use with the servo
  val = map(val, sensorMin, sensorMax, 0, 180);
  // sets the servo position according to the scaled value
  myservo.write(val);
  // waits for 400 milliseconds to avoid clutter or inaccurate readings from UltraSonic Distance Sensor
  delay(400);
}

3: Your input device is slightly noisy, leading the measurement to randomly deviate from the true measurement up or down by 10%. How can we address this? Answer in (pseudo)code.


          const int numReadings = 10;
          int readings[numReadings];
          int readIndex = 0;
          int total = 0;
          int average = 0;
          int sensorValue;
          int inputPin = A0;

          void setup() {
            for (int thisReading = 0; thisReading < numReadings; thisReading++) {
              readings[thisReading] = 0;
            }
          }
          void loop() {
            sensorValue = analogRead(inputPin);
            total = total - readings[readIndex];
            readings[readIndex] = sensorValue;
            total = total + readings[readIndex];
            readIndex = readIndex + 1;

            if (readIndex >= numReadings) {
              readIndex = 0;
            }

            average = total / numReadings;
          }
